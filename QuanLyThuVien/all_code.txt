// === MatKhau.cs ===
using System.Security.Cryptography;
using System.Text;

namespace QuanLyThuVien.Data;

public static class MatKhau
{
    private const string TienTo = "PBKDF2";
    private const int SoVong = 100_000;
    private const int DoDaiMuoi = 16;
    private const int DoDaiBam = 32;

    public static string Bam(string matKhau)
    {
        byte[] muoi = RandomNumberGenerator.GetBytes(DoDaiMuoi);
        byte[] bam = Rfc2898DeriveBytes.Pbkdf2(matKhau, muoi, SoVong, HashAlgorithmName.SHA256, DoDaiBam);
        return $"{TienTo}${SoVong}${Convert.ToBase64String(muoi)}${Convert.ToBase64String(bam)}";
    }

    public static bool Kiem(string matKhauNhap, string chuoiDaLuu)
    {
        if (string.IsNullOrEmpty(chuoiDaLuu)) return false;

        if (chuoiDaLuu.StartsWith(TienTo + "$", StringComparison.Ordinal))
        {
            string[] phan = chuoiDaLuu.Split('$');
            if (phan.Length != 4 || !int.TryParse(phan[1], out int soVong) || soVong < 1)
                return false;

            byte[] muoi, bamLuu;
            try
            {
                muoi = Convert.FromBase64String(phan[2]);
                bamLuu = Convert.FromBase64String(phan[3]);
            }
            catch (FormatException)
            {
                return false;
            }
            if (muoi.Length == 0 || bamLuu.Length == 0) return false;

            byte[] bamNhap = Rfc2898DeriveBytes.Pbkdf2(
                matKhauNhap, muoi, soVong, HashAlgorithmName.SHA256, bamLuu.Length);
            return CryptographicOperations.FixedTimeEquals(bamNhap, bamLuu);
        }

        // Định dạng cũ
        return string.Equals(BamCu(matKhauNhap), chuoiDaLuu, StringComparison.OrdinalIgnoreCase);
    }

    // true nếu chuỗi đã lưu còn là định dạng cũ, cần nâng cấp
    public static bool CanNangCap(string chuoiDaLuu)
        => !chuoiDaLuu.StartsWith(TienTo + "$", StringComparison.Ordinal);

    public const int DoDaiToiThieu = 6;
    public const int DoDaiToiDa = 64;

    private static readonly HashSet<string> MatKhauYeu = new(StringComparer.OrdinalIgnoreCase)
    {
        "123", "123456", "1234567", "12345678", "123456789", "111111", "000000",
        "password", "matkhau", "qwerty", "abc123", "admin123", "nlu123"
    };

    public static string? KiemTraDoManh(string matKhau, string? khongDuocTrung = null)
    {
        if (matKhau.Length < DoDaiToiThieu)
            return $"Mật khẩu cần ít nhất {DoDaiToiThieu} ký tự.";
        if (matKhau.Length > DoDaiToiDa)
            return $"Mật khẩu tối đa {DoDaiToiDa} ký tự.";
        if (MatKhauYeu.Contains(matKhau))
            return "Mật khẩu này quá dễ đoán, hãy chọn mật khẩu khác.";
        if (!matKhau.Any(char.IsLetter) || !matKhau.Any(char.IsDigit))
            return "Mật khẩu cần có cả chữ và số.";
        if (!string.IsNullOrEmpty(khongDuocTrung)
            && string.Equals(matKhau, khongDuocTrung, StringComparison.OrdinalIgnoreCase))
            return "Mật khẩu không được trùng với mã số sinh viên.";
        return null;
    }

    private static string BamCu(string matKhau)
    {
        byte[] bytes = SHA256.HashData(Encoding.UTF8.GetBytes("NLU_SALT_" + matKhau));
        return Convert.ToHexString(bytes);
    }
}

// === MuonService.cs ===
using Microsoft.EntityFrameworkCore;
using QuanLyThuVien.Entities;

namespace QuanLyThuVien.Data;

public enum LyDoChan { KhongCo, KhongTimThay, KhoaThe, NoPhat, DayGioiHan }

public sealed record KetQuaDieuKien(
    LyDoChan Chan, string MaDocGia, DocGia? DocGia, int DangGiu, decimal SoNo, int SoMuonThem)
{
    public bool Du => Chan == LyDoChan.KhongCo;


    public string MoTa => Chan switch
    {
        LyDoChan.KhongTimThay => $"Không tìm thấy độc giả có mã \"{MaDocGia}\".",
        LyDoChan.KhoaThe => $"Thẻ của {DocGia!.HoTen} đang bị khóa, không thể mượn sách.",
        LyDoChan.NoPhat => $"{DocGia!.HoTen} đang nợ phạt {SoNo:N0} VNĐ.\nYêu cầu đóng phạt trước (màn hình Quản lý Độc giả).",
        LyDoChan.DayGioiHan when SoMuonThem <= 1 =>
            $"{DocGia!.HoTen} đang giữ {DangGiu} cuốn chưa trả, đã đạt mức tối đa {MuonService.SoSachToiDa} cuốn.",
        LyDoChan.DayGioiHan =>
            $"Vượt quá giới hạn {MuonService.SoSachToiDa} cuốn: {DocGia!.HoTen} đang giữ {DangGiu} cuốn, muốn mượn thêm {SoMuonThem} cuốn.",
        _ => string.Empty
    };
}


public static class MuonService
{
    public const int SoSachToiDa = 3;   // tối đa 3 cuốn đang giữ
    public const int SoNgayMuon = 14;   // ngày phải trả = ngày mượn + 14

    public static Task<int> DemSachDangGiuAsync(AppDbContext db, string maDocGia)
        => db.ChiTietPhieuMuons.CountAsync(
            ct => ct.NgayTraThucTe == null && ct.PhieuMuon!.MaDocGia == maDocGia);

    /// <param name="soMuonThem">Số cuốn định mượn thêm (0 = chỉ kiểm tra còn được mượn tối thiểu 1 cuốn).</param>
    public static async Task<KetQuaDieuKien> KiemTraAsync(AppDbContext db, string maDocGia, int soMuonThem = 0)
    {
        var dg = await db.DocGias.AsNoTracking().FirstOrDefaultAsync(d => d.MaDocGia == maDocGia);
        if (dg == null)
            return new(LyDoChan.KhongTimThay, maDocGia, null, 0, 0, soMuonThem);

        int dangGiu = await DemSachDangGiuAsync(db, maDocGia);

        // Nợ = số lớn hơn giữa TienNoPhat và tổng hóa đơn chưa thanh toán (cộng trong bộ nhớ vì SQLite không SUM decimal)
        var hoaDon = await db.HoaDonPhats.AsNoTracking()
            .Where(h => h.MaDocGia == maDocGia && !h.DaThanhToan)
            .Select(h => new { h.TienPhatQuaHan, h.TienBoiThuong })
            .ToListAsync();
        decimal soNo = Math.Max(dg.TienNoPhat, hoaDon.Sum(h => h.TienPhatQuaHan + h.TienBoiThuong));

        if (dg.BiKhoaThe) return new(LyDoChan.KhoaThe, maDocGia, dg, dangGiu, soNo, soMuonThem);
        if (soNo > 0) return new(LyDoChan.NoPhat, maDocGia, dg, dangGiu, soNo, soMuonThem);
        if (dangGiu + Math.Max(1, soMuonThem) > SoSachToiDa)
            return new(LyDoChan.DayGioiHan, maDocGia, dg, dangGiu, soNo, soMuonThem);

        return new(LyDoChan.KhongCo, maDocGia, dg, dangGiu, soNo, soMuonThem);
    }
}

// === PhatService.cs ===
using Microsoft.EntityFrameworkCore;
using QuanLyThuVien.Entities;

namespace QuanLyThuVien.Data;

/// <summary>
/// Nơi DUY NHẤT được tạo / thu hóa đơn phạt và cập nhật DocGia.TienNoPhat.
/// Các hàm chỉ thay đổi trong DbContext, người gọi tự gọi SaveChanges
/// để hóa đơn + nợ cùng nằm trong một transaction.
/// </summary>
public static class PhatService
{
    // Tạo hóa đơn phạt. Chưa thanh toán thì cộng vào nợ của độc giả.
    public static async Task TaoHoaDonAsync(AppDbContext db, string maDocGia, int? maPhieuMuon,
        decimal tienPhatQuaHan, decimal tienBoiThuong, string? lyDo, bool daThanhToan)
    {
        decimal tong = tienPhatQuaHan + tienBoiThuong;
        if (tong <= 0) return;

        db.HoaDonPhats.Add(new HoaDonPhat
        {
            MaDocGia = maDocGia,
            MaPhieuMuon = maPhieuMuon,
            NgayLap = DateTime.Now,
            TienPhatQuaHan = tienPhatQuaHan,
            TienBoiThuong = tienBoiThuong,
            LyDo = string.IsNullOrWhiteSpace(lyDo) ? "Phạt vi phạm mượn trả" : lyDo,
            DaThanhToan = daThanhToan
        });

        if (!daThanhToan)
        {
            var dg = await db.DocGias.FirstOrDefaultAsync(d => d.MaDocGia == maDocGia);
            if (dg != null) dg.TienNoPhat += tong;
        }
    }

    // Thu tiền một hóa đơn. Trả false nếu hóa đơn không tồn tại hoặc đã thanh toán.
    public static async Task<bool> ThuHoaDonAsync(AppDbContext db, int maHoaDon)
    {
        var hd = await db.HoaDonPhats.FirstOrDefaultAsync(h => h.MaHoaDon == maHoaDon);
        if (hd == null || hd.DaThanhToan) return false;

        hd.DaThanhToan = true;

        var dg = await db.DocGias.FirstOrDefaultAsync(d => d.MaDocGia == hd.MaDocGia);
        if (dg != null)
            dg.TienNoPhat = await TinhNoConLaiAsync(db, hd.MaDocGia, boQuaHoaDon: hd.MaHoaDon);

        return true;
    }

    // Thu toàn bộ hóa đơn chưa thanh toán của một độc giả. Trả về tổng số tiền đã thu.
    public static async Task<decimal> ThuTatCaAsync(AppDbContext db, string maDocGia)
    {
        var dg = await db.DocGias.FirstOrDefaultAsync(d => d.MaDocGia == maDocGia);
        if (dg == null) return 0;

        var hoaDons = await db.HoaDonPhats
            .Where(h => h.MaDocGia == maDocGia && !h.DaThanhToan)
            .ToListAsync();
        foreach (var h in hoaDons) h.DaThanhToan = true;

        decimal daThu = dg.TienNoPhat;
        dg.TienNoPhat = 0;
        return daThu;
    }

    // Nợ còn lại = tổng hóa đơn chưa thanh toán 
    private static async Task<decimal> TinhNoConLaiAsync(AppDbContext db, string maDocGia, int boQuaHoaDon)
    {
        var ds = await db.HoaDonPhats.AsNoTracking()
            .Where(h => h.MaDocGia == maDocGia && !h.DaThanhToan && h.MaHoaDon != boQuaHoaDon)
            .Select(h => new { h.TienPhatQuaHan, h.TienBoiThuong })
            .ToListAsync();
        return ds.Sum(x => x.TienPhatQuaHan + x.TienBoiThuong);
    }
}

    public static void KhoiTaoDuLieuTaiKhoan(AppDbContext db)

                ""MaTaiKhoan"" INTEGER NOT NULL CONSTRAINT ""PK_TaiKhoans"" PRIMARY KEY AUTOINCREMENT,

                ""VaiTro"" TEXT NOT NULL,
                ""MaDocGia"" TEXT NULL,
                ""BatBuocDoiMatKhau"" INTEGER NOT NULL DEFAULT 0
            );");
        db.Database.ExecuteSqlRaw(@"
            CREATE TABLE IF NOT EXISTS ""YeuCauMuons"" (
                ""MaYeuCau"" INTEGER NOT NULL CONSTRAINT ""PK_YeuCauMuons"" PRIMARY KEY AUTOINCREMENT,
                ""MaDocGia"" TEXT NOT NULL,
                ""MaDauSach"" TEXT NOT NULL,
                ""NgayGui"" TEXT NOT NULL,
                ""TrangThai"" TEXT NOT NULL,
                CONSTRAINT ""FK_YeuCauMuons_DocGias"" FOREIGN KEY (""MaDocGia"") REFERENCES ""DocGias"" (""MaDocGia"") ON DELETE RESTRICT,
                CONSTRAINT ""FK_YeuCauMuons_DauSachs"" FOREIGN KEY (""MaDauSach"") REFERENCES ""DauSachs"" (""MaDauSach"") ON DELETE RESTRICT
            );");

        TaoChiMucDuyNhat(db);

        // Tài khoản Thủ thư mặc định 
        if (!db.TaiKhoans.Any(t => t.VaiTro == LoaiTaiKhoan.ThuThu))
        {
            string matKhauMacDinh = MatKhau.Bam("123");
            db.TaiKhoans.AddRange(
                new TaiKhoan
                {

                    HoTen = "Quản trị viên Thư viện",
                    VaiTro = LoaiTaiKhoan.ThuThu,
                    MaDocGia = null,
                    BatBuocDoiMatKhau = false
                },
                new TaiKhoan
                {
                    TenDangNhap = "thuthu@hcmuaf.edu.vn",
                    MatKhauHash = matKhauMacDinh,
                    HoTen = "Thủ thư trực quầy",
                    VaiTro = LoaiTaiKhoan.ThuThu,
                    MaDocGia = null,
                    BatBuocDoiMatKhau = false
                });
            db.SaveChanges();
        }

        DongBoTaiKhoanDocGia(db);
    }

    // Tạo tài khoản cho độc giả chưa có
    public static void DongBoTaiKhoanDocGia(AppDbContext db)
    {
        var docGiaThieuTk = db.DocGias.AsNoTracking()
            .Where(d => !db.TaiKhoans.Any(t => t.MaDocGia == d.MaDocGia))
            .ToList();
        if (docGiaThieuTk.Count == 0) return;

        var tenDaDung = db.TaiKhoans.AsNoTracking()
            .Select(t => t.TenDangNhap.ToLower())
            .ToHashSet();

        string matKhauMacDinh = MatKhau.Bam("123");
        foreach (var dg in docGiaThieuTk)
        {
            string email = $"{dg.MaDocGia.ToLowerInvariant()}@st.hcmuaf.edu.vn";
            if (!tenDaDung.Add(email)) continue;   // email đã bị tài khoản khác dùng

            db.TaiKhoans.Add(new TaiKhoan
            {
                TenDangNhap = email,
                MatKhauHash = matKhauMacDinh,
                HoTen = dg.HoTen,
                VaiTro = LoaiTaiKhoan.DocGia,
                MaDocGia = dg.MaDocGia,
                BatBuocDoiMatKhau = true
            });
        }
        db.SaveChanges();
    }

    // Chặn trùng tên đăng nhập
    private static void TaoChiMucDuyNhat(AppDbContext db)
    {
        try
        {
            db.Database.ExecuteSqlRaw(
                @"CREATE UNIQUE INDEX IF NOT EXISTS ""IX_TaiKhoans_TenDangNhap"" ON ""TaiKhoans"" (""TenDangNhap"" COLLATE NOCASE);");
            db.Database.ExecuteSqlRaw(
                @"CREATE UNIQUE INDEX IF NOT EXISTS ""IX_TaiKhoans_MaDocGia"" ON ""TaiKhoans"" (""MaDocGia"");");
        }
        catch (Exception ex)
        {
            NhatKy.Ghi(ex);   
        }
    }
}

/// <summary>
/// Sinh viên tự đăng ký lần đầu: tạo cùng lúc 1 DocGia và 1 TaiKhoan.
/// FrmDangNhap đọc MaSinhVien để điền sẵn ô đăng nhập.
/// </summary>
public partial class DlgDangKy : Form
{
    public string MaSinhVien { get; private set; } = string.Empty;

    public DlgDangKy()
    {
        InitializeComponent();
        Theme.Apply(this);

        lblTieuDe.Font = Theme.FontTitle;
        lblTieuDe.ForeColor = Theme.PrimaryDark;
        lblGhiChu.ForeColor = Theme.Neutral;
        Theme.StyleButton(btnHuy, Theme.Neutral);
        txtMatKhau.MaxLength = MatKhau.DoDaiToiDa;
        txtXacNhan.MaxLength = MatKhau.DoDaiToiDa;
    }

    private void chkHienMatKhau_CheckedChanged(object? sender, EventArgs e)
    {
        bool hien = chkHienMatKhau.Checked;
        txtMatKhau.UseSystemPasswordChar = !hien;
        txtXacNhan.UseSystemPasswordChar = !hien;
    }

    private void BaoLoi(string noiDung, Control oCanSua)
    {
        Toast.CanhBao(this, noiDung);
        oCanSua.Focus();
    }

    private async void btnDangKy_Click(object? sender, EventArgs e)
    {
        string ma = txtMaSv.Text.Trim();
        string hoTen = Regex.Replace(txtHoTen.Text.Trim(), @"\s+", " ");
        string sdt = txtSdt.Text.Trim();
        string matKhau = txtMatKhau.Text;

        // ===== Kiểm tra dữ liệu nhập =====
        if (!Regex.IsMatch(ma, @"^\d{8}$"))
        { BaoLoi("Mã số sinh viên gồm đúng 8 chữ số (ví dụ 24130265).", txtMaSv); return; }
        if (hoTen.Split(' ').Length < 2)
        { BaoLoi("Hãy nhập đầy đủ họ và tên (ít nhất 2 từ).", txtHoTen); return; }
        if (!Regex.IsMatch(sdt, @"^0\d{9}$"))
        { BaoLoi("Số điện thoại gồm 10 chữ số và bắt đầu bằng 0.", txtSdt); return; }

        string? loiMatKhau = MatKhau.KiemTraDoManh(matKhau, ma);
        if (loiMatKhau != null)
        { BaoLoi(loiMatKhau, txtMatKhau); return; }
        if (matKhau != txtXacNhan.Text)
        { BaoLoi("Hai mật khẩu không khớp.", txtXacNhan); return; }

        btnDangKy.Enabled = false;
        try
        {
            using var db = new AppDbContext();
            string email = $"{ma}@st.hcmuaf.edu.vn";

            // Mã đã có (do thủ thư tạo hoặc đã đăng ký) thì không cho đăng ký lại, 
            bool daCo = await db.DocGias.AnyAsync(d => d.MaDocGia == ma)
                     || await db.TaiKhoans.AnyAsync(t => t.MaDocGia == ma || t.TenDangNhap == email);
            if (daCo)
            {
                BaoLoi("Mã số sinh viên này đã có trong hệ thống. Hãy đăng nhập, hoặc liên hệ thủ thư nếu bạn quên mật khẩu.", txtMaSv);
                return;
            }

            // Lưu cả hai bản ghi trong một lần SaveChanges để không bị tạo dở dang
            db.DocGias.Add(new DocGia
            {
                MaDocGia = ma,
                HoTen = hoTen,
                SoDienThoai = sdt,
                BiKhoaThe = false,
                TienNoPhat = 0
            });
            db.TaiKhoans.Add(new TaiKhoan
            {
                TenDangNhap = email,
                MatKhauHash = MatKhau.Bam(matKhau),   // luôn băm, không lưu mật khẩu thô
                HoTen = hoTen,
                VaiTro = "DocGia",
                MaDocGia = ma,
                BatBuocDoiMatKhau = false             // đã tự đặt mật khẩu nên không bắt đổi nữa
            });
            await db.SaveChangesAsync();

            MaSinhVien = ma;
            DialogResult = DialogResult.OK;
        }
        catch (Exception ex)
        {
            MessageBox.Show($"Không thể đăng ký:\n{ex.Message}", "Lỗi",
                MessageBoxButtons.OK, MessageBoxIcon.Error);
        }
        finally
        {
            btnDangKy.Enabled = true;
        }
    }
}

    private async Task NapDanhSachAsync()
    {
        using var db = new AppDbContext();
        var ds = await db.YeuCauMuons.AsNoTracking()
            .Include(y => y.DocGia)
            .Include(y => y.DauSach).ThenInclude(d => d!.CuonSachs)
            .Where(y => y.TrangThai == TrangThaiYeuCau.ChoDuyet)
            .OrderBy(y => y.NgayGui)
            .ToListAsync();

        var dong = ds.Select(y => new DongYeuCau(
                y.MaYeuCau,
                y.MaDocGia,
                y.DocGia?.HoTen ?? "",
                y.MaDauSach,
                y.DauSach?.TenDauSach ?? "",
                y.NgayGui.ToString("dd/MM/yyyy HH:mm"),
                y.DauSach?.CuonSachs.Count(c => c.TrangThai == TrangThaiSach.SanSang) ?? 0))
            .ToList();

        dgvYeuCau.DataSource = dong;
        lblDem.Text = $"Có {dong.Count} yêu cầu chờ duyệt";
        btnDuyet.Enabled = dong.Count > 0;
        btnTuChoi.Enabled = dong.Count > 0;
    }

    // Chỉ KIỂM TRA và trả kết quả cho FrmMuonSach. Yêu cầu chỉ thành "Đã duyệt" khi phiếu mượn được lập xong.
    private async void btnDuyet_Click(object? sender, EventArgs e)
    {
        try
        {
            if (dgvYeuCau.CurrentRow?.DataBoundItem is not DongYeuCau dong)
            {
                Toast.CanhBao(this, "Hãy chọn một yêu cầu cần duyệt.");
                return;
            }

            using var db = new AppDbContext();

            // Đọc lại từ CSDL: có thể yêu cầu đã được xử lý ở nơi khác
            var yc = await db.YeuCauMuons.AsNoTracking()
                .FirstOrDefaultAsync(y => y.MaYeuCau == dong.MaYeuCau);
            if (yc == null || yc.TrangThai != TrangThaiYeuCau.ChoDuyet)
            {
                Toast.CanhBao(this, "Yêu cầu này đã được xử lý trước đó.");
                await NapDanhSachAsync();
                return;
            }

            // Cùng quy tắc với màn hình Mượn sách: khóa thẻ, nợ phạt, đủ 3 cuốn
            var kq = await MuonService.KiemTraAsync(db, yc.MaDocGia);
            if (!kq.Du)
            {
                Toast.Loi(this, kq.MoTa + "\nNếu không thể xử lý, hãy chọn Từ chối.");
                return;
            }

            var cuon = await db.CuonSachs.AsNoTracking()
                .Where(c => c.MaDauSach == yc.MaDauSach && c.TrangThai == TrangThaiSach.SanSang)
                .OrderBy(c => c.MaCuonSach)
                .FirstOrDefaultAsync();
            if (cuon == null)
            {
                Toast.Loi(this, "Đầu sách này hiện không còn bản sẵn sàng.");
                await NapDanhSachAsync();
                return;
            }

            MaYeuCauDuocChon = yc.MaYeuCau;
            MaDocGiaDuocChon = yc.MaDocGia;
            MaDauSachDuocChon = yc.MaDauSach;
            MaCuonSachGoiY = cuon.MaCuonSach;
            DialogResult = DialogResult.OK;       
        }
        catch (Exception ex)
        {
            NhatKy.Ghi(ex);
            Toast.Loi(this, "Không thể xử lý yêu cầu. Vui lòng thử lại.");
        }
    }

    // ===== Từ chối =====
    private async void btnTuChoi_Click(object? sender, EventArgs e)
    {
        if (dgvYeuCau.CurrentRow?.DataBoundItem is not DongYeuCau dong)
        {
            Toast.CanhBao(this, "Hãy chọn một yêu cầu cần từ chối.");
            return;
        }

        var hoi = MessageBox.Show(
            $"Từ chối yêu cầu mượn \"{dong.TenDauSach}\" của {dong.HoTen}?",
            "Xác nhận", MessageBoxButtons.YesNo, MessageBoxIcon.Question);
        if (hoi != DialogResult.Yes) return;

        using var db = new AppDbContext();
        var yc = await db.YeuCauMuons.FindAsync(dong.MaYeuCau);
        if (yc == null || yc.TrangThai != TrangThaiYeuCau.ChoDuyet)
        {
            Toast.CanhBao(this, "Yêu cầu này đã được xử lý trước đó.");
            await NapDanhSachAsync();
            return;
        }

        yc.TrangThai = TrangThaiYeuCau.TuChoi;
        await db.SaveChangesAsync();

        Toast.ThanhCong(this, $"Đã từ chối yêu cầu của {dong.HoTen}.");
        await NapDanhSachAsync();
    }

    private void btnXuatCsv_Click(object? sender, EventArgs e)
    {
        var (gridHienTai, tenFile) = LayLuoiDangHienThi();

        if (gridHienTai.Rows.Count == 0)
        {
            Toast.CanhBao(this, "Không có dữ liệu để xuất trong tab hiện tại.");
            return;
        }

        using var sfd = new SaveFileDialog
        {
            Filter = "CSV file (*.csv)|*.csv",
            FileName = $"BaoCao_{tenFile}_{DateTime.Now:yyyyMMdd_HHmm}.csv"
        };

        if (sfd.ShowDialog(this) != DialogResult.OK)
            return;

        try
        {
            var sb = new StringBuilder();
            var visibleCols = gridHienTai.Columns
                .Cast<DataGridViewColumn>()
                .Where(c => c.Visible)
                .OrderBy(c => c.DisplayIndex)
                .ToList();

            sb.AppendLine(string.Join(",",
                visibleCols.Select(c => $"\"{c.HeaderText.Replace("\"", "\"\"")}\"")));

            foreach (DataGridViewRow row in gridHienTai.Rows)
            {
                if (row.IsNewRow) continue;

                var cells = visibleCols.Select(col =>
                {
                    string giaTri = row.Cells[col.Index].FormattedValue?.ToString() ?? "";
                    return $"\"{giaTri.Replace("\"", "\"\"")}\"";
                });
                sb.AppendLine(string.Join(",", cells));
            }

            // UTF-8 có BOM để Microsoft Excel đọc chuẩn tiếng Việt có dấu
            var utf8WithBom = new UTF8Encoding(encoderShouldEmitUTF8Identifier: true);
            File.WriteAllText(sfd.FileName, sb.ToString(), utf8WithBom);

            Toast.ThanhCong(this, "Xuất báo cáo CSV thành công!");
        }
        catch (IOException ex)
        {
            Toast.Loi(this, $"Không thể ghi file (có thể file đang mở ở Excel): {ex.Message}");
        }
        catch (Exception ex)
        {
            Toast.Loi(this, $"Xuất CSV thất bại: {ex.Message}");
        }
    }

    // =====================================================================
    // NẠP DỮ LIỆU BÁO CÁO (ASYNC)
    // =====================================================================
    private async Task NapDuLieuBaoCaoAsync()
    {
        if (_dangTai) return;
        _dangTai = true;
        DatTrangThaiDieuKhien(false);

        try
        {
            using var context = new AppDbContext();

            DateTime tuNgay = dtpTuNgay.Value.Date;
            DateTime denNgay = dtpDenNgay.Value.Date.AddDays(1).AddTicks(-1);
            DateTime homNay = DateTime.Today;

            // 1. Báo cáo Mượn - Trả trong kỳ lọc
            var dsMuonTraRaw = await context.ChiTietPhieuMuons
                .AsNoTracking()
                .Where(ct => ct.PhieuMuon != null
                          && ct.PhieuMuon.NgayMuon >= tuNgay
                          && ct.PhieuMuon.NgayMuon <= denNgay)
                .OrderByDescending(ct => ct.PhieuMuon!.NgayMuon)
                .Select(ct => new
                {
                    MaPhieu = ct.MaPhieuMuon,
                    TenDocGia = ct.PhieuMuon!.DocGia != null ? ct.PhieuMuon.DocGia.HoTen : "",
                    TenDauSach = ct.CuonSach != null && ct.CuonSach.DauSach != null
                        ? ct.CuonSach.DauSach.TenDauSach
                        : "",
                    ct.MaCuonSach,
                    ct.PhieuMuon.NgayMuon,
                    HanTra = ct.PhieuMuon.NgayPhaiTra,
                    ct.NgayTraThucTe
                })
                .ToListAsync();

            var dsMuonTra = dsMuonTraRaw.Select(ct => new BaoCaoMuonTraDto
            {
                MaPhieu = ct.MaPhieu,
                TenDocGia = ct.TenDocGia ?? "",
                TenDauSach = ct.TenDauSach ?? "",
                MaCuonSach = ct.MaCuonSach,
                NgayMuon = ct.NgayMuon,
                HanTra = ct.HanTra,
                NgayTraThucTe = ct.NgayTraThucTe,
                TrangThai = ct.NgayTraThucTe != null
                    ? "Đã trả"
                    : (homNay > ct.HanTra.Date ? "Quá hạn" : "Đang mượn")
            }).ToList();

            // 2. Biểu đồ 5 tháng gần nhất tính đến tháng của dtpDenNgay
            var thangKetThuc = new DateTime(dtpDenNgay.Value.Year, dtpDenNgay.Value.Month, 1);
            var thangBatDau = thangKetThuc.AddMonths(-4);
            var ngayCuoiBieuDo = thangKetThuc.AddMonths(1).AddTicks(-1);

            var ngayMuon5Thang = await context.ChiTietPhieuMuons
                .AsNoTracking()
                .Where(ct => ct.PhieuMuon != null
                          && ct.PhieuMuon.NgayMuon >= thangBatDau
                          && ct.PhieuMuon.NgayMuon <= ngayCuoiBieuDo)
                .Select(ct => ct.PhieuMuon!.NgayMuon)
                .ToListAsync();

            var soLuotTheoThang = ngayMuon5Thang
                .GroupBy(d => new DateTime(d.Year, d.Month, 1))
                .ToDictionary(g => g.Key, g => g.Count());

            var duLieuBieuDoMoi = new List<KeyValuePair<string, int>>(5);
            for (int i = 4; i >= 0; i--)
            {
                var thang = thangKetThuc.AddMonths(-i);
                soLuotTheoThang.TryGetValue(thang, out int soLuot);
                duLieuBieuDoMoi.Add(new KeyValuePair<string, int>($"T{thang:MM/yy}", soLuot));
            }

            // 3. Top Đầu Sách Mượn Nhiều trong kỳ lọc
            var tongSoCuonTheoDauSach = await context.CuonSachs
                .AsNoTracking()
                .GroupBy(c => c.MaDauSach)
                .Select(g => new { MaDauSach = g.Key, Tong = g.Count() })
                .ToDictionaryAsync(x => x.MaDauSach, x => x.Tong);

            var topSachRaw = await context.ChiTietPhieuMuons
                .AsNoTracking()
                .Where(ct => ct.PhieuMuon != null
                          && ct.CuonSach != null
                          && ct.CuonSach.DauSach != null
                          && ct.PhieuMuon.NgayMuon >= tuNgay
                          && ct.PhieuMuon.NgayMuon <= denNgay)
                .GroupBy(ct => new
                {
                    ct.CuonSach!.DauSach!.MaDauSach,
                    ct.CuonSach.DauSach.TenDauSach,
                    TenTheLoai = ct.CuonSach.DauSach.TheLoai != null
                        ? ct.CuonSach.DauSach.TheLoai.TenTheLoai
                        : ""
                })
                .Select(g => new
                {
                    g.Key.MaDauSach,
                    TenSach = g.Key.TenDauSach ?? "",
                    TheLoai = g.Key.TenTheLoai ?? "",
                    SoLuotMuon = g.Count()
                })
                .OrderByDescending(x => x.SoLuotMuon)
                .ToListAsync();

            var topSach = topSachRaw.Select(x => new TopDauSachDto
            {
                MaDauSach = x.MaDauSach,
                TenSach = x.TenSach,
                TheLoai = x.TheLoai,
                SoLuotMuon = x.SoLuotMuon,
                TongSoCuon = tongSoCuonTheoDauSach.TryGetValue(x.MaDauSach, out int tong) ? tong : 0
            }).ToList();

            // 4. Thống kê Hóa Đơn Phạt trong kỳ lọc
            var dsPhat = await context.HoaDonPhats
                .AsNoTracking()
                .Where(h => h.NgayLap >= tuNgay && h.NgayLap <= denNgay)
                .OrderByDescending(h => h.NgayLap)
                .Select(h => new BaoCaoTienPhatDto
                {
                    MaHoaDon = h.MaHoaDon,
                    TenDocGia = h.DocGia != null ? (h.DocGia.HoTen ?? "") : "",
                    NgayLap = h.NgayLap,
                    SoTien = h.TienPhatQuaHan + h.TienBoiThuong,
                    LyDo = h.LyDo ?? "",
                    TrangThai = h.DaThanhToan ? "Đã thanh toán" : "Chưa thanh toán"
                })
                .ToListAsync();

            // 5. Độc giả quá hạn chưa trả (toàn hệ thống)
            var dsDocGiaQuaHanRaw = await context.ChiTietPhieuMuons
                .AsNoTracking()
                .Where(ct => ct.NgayTraThucTe == null
                          && ct.PhieuMuon != null
                          && ct.PhieuMuon.NgayPhaiTra < homNay
                          && ct.PhieuMuon.DocGia != null
                          && ct.CuonSach != null
                          && ct.CuonSach.DauSach != null)
                .OrderBy(ct => ct.PhieuMuon!.NgayPhaiTra)
                .Select(ct => new
                {
                    ct.PhieuMuon!.MaDocGia,
                    HoTen = ct.PhieuMuon.DocGia!.HoTen,
                    SoDienThoai = ct.PhieuMuon.DocGia.SoDienThoai,
                    TenDauSach = ct.CuonSach!.DauSach!.TenDauSach,
                    HanTra = ct.PhieuMuon.NgayPhaiTra
                })
                .ToListAsync();

            var dsDocGiaQuaHan = dsDocGiaQuaHanRaw
                .Select(x => new
                {
                    x.MaDocGia,
                    HoTen = x.HoTen ?? "",
                    SoDienThoai = x.SoDienThoai ?? "",
                    TenDauSach = x.TenDauSach ?? "",
                    x.HanTra,
                    SoNgayTre = (homNay - x.HanTra.Date).Days
                })
                .ToList();

            // 6. Chỉ số KPI toàn hệ thống
            int dangMuon = await context.ChiTietPhieuMuons
                .AsNoTracking()
                .CountAsync(ct => ct.NgayTraThucTe == null);

            int quaHan = await context.ChiTietPhieuMuons
                .AsNoTracking()
                .CountAsync(ct => ct.NgayTraThucTe == null
                               && ct.PhieuMuon != null
                               && ct.PhieuMuon.NgayPhaiTra < homNay);

            decimal tongThuPhat = dsPhat
                .Where(x => x.TrangThai == "Đã thanh toán")
                .Sum(x => x.SoTien);

            if (IsDisposed) return;

            // 7. Đổ dữ liệu lên giao diện
            dgvMuonTra.DataSource = dsMuonTra;
            dgvTopSach.DataSource = topSach;
            dgvTienPhat.DataSource = dsPhat;
            dgvQuaHan.DataSource = dsDocGiaQuaHan;

            dgvMuonTra.Invalidate();
            dgvTopSach.Invalidate();
            dgvTienPhat.Invalidate();
            dgvQuaHan.Invalidate();

            _duLieuBieuDoMuon.Clear();
            _duLieuBieuDoMuon.AddRange(duLieuBieuDoMoi);
            pnlBieuDoMuon.Invalidate();

            lblKpiLuotMuon.Text = dsMuonTra.Count.ToString("N0");
            lblKpiDangMuon.Text = dangMuon.ToString("N0");
            lblKpiQuaHan.Text = quaHan.ToString("N0");

            string textTienPhat = $"{tongThuPhat:N0} VNĐ";
            lblKpiTienPhat.Font = textTienPhat.Length > 11 ? _fontKpiValueCompact : _fontKpiValue;
            lblKpiTienPhat.Text = textTienPhat;
        }
        catch (Exception ex)
        {
            NhatKy.Ghi(ex);
            if (!IsDisposed)
            {
                KhoiTaoKhungBieuDoMacDinh(dtpDenNgay.Value);
                pnlBieuDoMuon.Invalidate();
#if DEBUG
                MessageBox.Show(this, ex.ToString(), "Chi tiết lỗi tải báo cáo",
                    MessageBoxButtons.OK, MessageBoxIcon.Error);
#else
                Toast.Loi(this, "Không thể tải báo cáo. Chi tiết đã được ghi vào error.log.");
#endif
            }
        }
        finally
        {
            _dangTai = false;
            if (!IsDisposed)
                DatTrangThaiDieuKhien(true);
        }
    }

    private void KhoiTaoKhungBieuDoMacDinh(DateTime mocDenNgay)
    {
        _duLieuBieuDoMuon.Clear();
        var thangKetThuc = new DateTime(mocDenNgay.Year, mocDenNgay.Month, 1);
        for (int i = 4; i >= 0; i--)

    private void pnlBieuDoMuon_Paint(object? sender, PaintEventArgs e)
    {
        var bounds = pnlBieuDoMuon.ClientRectangle;
        if (bounds.Width < 120 || bounds.Height < 150)
            return;

        var g = e.Graphics;
        g.SmoothingMode = SmoothingMode.AntiAlias;
        g.Clear(pnlBieuDoMuon.BackColor);

        using (var cardBorder = new Pen(Theme.Border))
            g.DrawRectangle(cardBorder, 0, 0, bounds.Width - 1, bounds.Height - 1);

        using var titleFont = new Font("Segoe UI", 9F, FontStyle.Bold);
        using var subFont = new Font("Segoe UI", 8F, FontStyle.Regular);
        using var labelFont = new Font("Segoe UI", 8F, FontStyle.Regular);
        using var valFont = new Font("Segoe UI", 8.5F, FontStyle.Bold);

        using var titleBrush = new SolidBrush(Theme.PrimaryDark);
        using var subBrush = new SolidBrush(Theme.Neutral);
        using var labelBrush = new SolidBrush(Theme.TextDark);
        using var gridPen = new Pen(Color.FromArgb(226, 232, 240)) { DashStyle = DashStyle.Dash };
        using var axisPen = new Pen(Color.FromArgb(148, 163, 184));
        using var barBrush = new SolidBrush(Color.FromArgb(45, 160, 150));
        using var barLatestBrush = new SolidBrush(Theme.Primary);
        using var zeroBrush = new SolidBrush(Color.FromArgb(203, 213, 225));

        using var ellipsis = new StringFormat
        {
            Trimming = StringTrimming.EllipsisCharacter,
            FormatFlags = StringFormatFlags.NoWrap
        };

        g.DrawString(
            "LƯỢT MƯỢN 5 THÁNG GẦN NHẤT",
            titleFont,
            titleBrush,
            new RectangleF(12, 12, bounds.Width - 24, 20),
            ellipsis);

        g.DrawString(
            $"Mốc tính đến tháng {dtpDenNgay.Value:MM/yyyy}",
            subFont,
            subBrush,
            new RectangleF(12, 32, bounds.Width - 24, 16),
            ellipsis);

        var plot = new Rectangle(38, 60, bounds.Width - 52, bounds.Height - 98);
        if (plot.Width <= 20 || plot.Height <= 40)
            return;

        int maxVal = Math.Max(4, _duLieuBieuDoMuon.Select(x => x.Value).DefaultIfEmpty().Max());
        int mocTren = Math.Max(4, (int)Math.Ceiling(maxVal / 4d) * 4);

        using var centered = new StringFormat { Alignment = StringAlignment.Center };
        using var rightAlign = new StringFormat { Alignment = StringAlignment.Far, LineAlignment = StringAlignment.Center };

        for (int i = 0; i <= 4; i++)
        {
            int giaTri = mocTren * i / 4;
            float y = plot.Bottom - plot.Height * i / 4f;
            g.DrawLine(i == 0 ? axisPen : gridPen, plot.Left, y, plot.Right, y);
            g.DrawString(
                giaTri.ToString("N0"),
                labelFont,
                subBrush,
                new RectangleF(2, y - 9, 30, 18),
                rightAlign);
        }

        if (_duLieuBieuDoMuon.Count == 0)
            return;

        float slotWidth = plot.Width / (float)_duLieuBieuDoMuon.Count;
        float barWidth = Math.Min(38f, slotWidth * 0.54f);

        for (int i = 0; i < _duLieuBieuDoMuon.Count; i++)
        {
            var item = _duLieuBieuDoMuon[i];
            float x = plot.Left + slotWidth * i + (slotWidth - barWidth) / 2f;
            bool laThangCuoi = i == _duLieuBieuDoMuon.Count - 1;

            if (item.Value > 0)
            {
                float barHeight = Math.Max(6f, plot.Height * item.Value / (float)mocTren);
                float y = plot.Bottom - barHeight;

                using var path = TaoCotBoGocTren(new RectangleF(x, y, barWidth, barHeight), 5f);
                g.FillPath(laThangCuoi ? barLatestBrush : barBrush, path);

                g.DrawString(
                    item.Value.ToString("N0"),
                    valFont,
                    laThangCuoi ? titleBrush : labelBrush,
                    new RectangleF(x - 10, Math.Max(plot.Top - 16, y - 20), barWidth + 20, 18),
                    centered);
            }
            else
            {
                g.FillRectangle(zeroBrush, x, plot.Bottom - 2, barWidth, 2);
                g.DrawString(
                    "0",
                    labelFont,
                    subBrush,
                    new RectangleF(x - 8, plot.Bottom - 20, barWidth + 16, 16),
                    centered);
            }

            g.DrawString(
                item.Key,
                laThangCuoi ? valFont : labelFont,
                laThangCuoi ? titleBrush : subBrush,
                new RectangleF(plot.Left + slotWidth * i, plot.Bottom + 8, slotWidth, 20),
                centered);
        }
    }

    private static GraphicsPath TaoCotBoGocTren(RectangleF r, float radius)
    {
        float d = Math.Min(radius * 2f, Math.Min(r.Width, r.Height));
        var path = new GraphicsPath();
        path.AddArc(r.X, r.Y, d, d, 180, 90);
        path.AddArc(r.Right - d, r.Y, d, d, 270, 90);
        path.AddLine(r.Right, r.Bottom, r.X, r.Bottom);
        path.CloseFigure();
        return path;
    }

    private void CauHinhLuoi(DataGridView dgv)
    {
        dgv.Dock = DockStyle.Fill;
        dgv.BackgroundColor = Theme.CardBg;
        dgv.BorderStyle = BorderStyle.None;
        dgv.AllowUserToAddRows = false;
        dgv.AllowUserToDeleteRows = false;
        dgv.ReadOnly = true;
        dgv.MultiSelect = false;
        dgv.RowHeadersVisible = false;
        dgv.SelectionMode = DataGridViewSelectionMode.FullRowSelect;
        dgv.AutoSizeColumnsMode = DataGridViewAutoSizeColumnsMode.Fill;

        dgv.ColumnHeadersHeightSizeMode = DataGridViewColumnHeadersHeightSizeMode.DisableResizing;
        dgv.ColumnHeadersHeight = 42;
        dgv.ColumnHeadersDefaultCellStyle.WrapMode = DataGridViewTriState.True;
        dgv.RowTemplate.Height = 34;

        dgv.DataBindingComplete += (_, _) =>
        {
            
            var cheDoCu = dgv.AutoSizeColumnsMode;
            dgv.AutoSizeColumnsMode = DataGridViewAutoSizeColumnsMode.None;
            try
            {
                foreach (DataGridViewColumn c in dgv.Columns)
                {
                    string prop = string.IsNullOrEmpty(c.DataPropertyName) ? c.Name : c.DataPropertyName;

                    if (CauHinhCot.TryGetValue(prop, out var cfg))
                    {
                        c.HeaderText = cfg.Header;
                        c.FillWeight = Math.Max(1F, cfg.Weight);
                        c.MinimumWidth = cfg.MinWidth > 0 ? cfg.MinWidth : 80;
                    }
                    else
                    {
                        c.MinimumWidth = 80;
                    }

                    if (c.ValueType == typeof(DateTime) || c.ValueType == typeof(DateTime?))
                    {
                        c.DefaultCellStyle.Format = "dd/MM/yyyy";
                        c.DefaultCellStyle.Alignment = DataGridViewContentAlignment.MiddleCenter;
                    }
                    else if (CotCanPhai.Contains(prop))
                    {
                        c.DefaultCellStyle.Format = "N0";
                        c.DefaultCellStyle.Alignment = DataGridViewContentAlignment.MiddleRight;
                    }
                    else if (CotCanGiua.Contains(prop))
                    {
                        c.DefaultCellStyle.Alignment = DataGridViewContentAlignment.MiddleCenter;
                    }
                }
            }
            finally
            {
                dgv.AutoSizeColumnsMode = cheDoCu;
            }
        };

        dgv.CellFormatting += (_, e) =>
        {
            if (e.RowIndex < 0 || e.ColumnIndex < 0 || e.CellStyle == null) return;
            string prop = dgv.Columns[e.ColumnIndex].DataPropertyName;

            if (prop == "TrangThai" && e.Value is string trangThai)
            {
                e.CellStyle.Font = Theme.FontBold;
                e.CellStyle.ForeColor = trangThai switch
                {
                    "Quá hạn" or "Chưa thanh toán" => Theme.Danger,
                    "Đã trả" or "Đã thanh toán"    => Color.FromArgb(22, 163, 74),
                    "Đang mượn"                    => Theme.Secondary,
                    _                              => Theme.TextDark
                };
            }
            else if (prop == "SoNgayTre" && e.Value is int ngayTre && ngayTre > 0)
            {
                e.CellStyle.Font = Theme.FontBold;
                e.CellStyle.ForeColor = Theme.Danger;
            }
        };

        dgv.Paint += (_, e) =>
        {
            if (dgv.Rows.Count > 0) return;

            int headerH = dgv.ColumnHeadersVisible && dgv.Columns.Count > 0 ? dgv.ColumnHeadersHeight : 0;
            int bodyH = Math.Max(40, dgv.ClientSize.Height - headerH);
            var rect = new RectangleF(16, headerH, Math.Max(100, dgv.ClientSize.Width - 32), bodyH);

            using var brush = new SolidBrush(Color.FromArgb(148, 163, 184));
            using var sf = new StringFormat
            {
                Alignment = StringAlignment.Center,
                LineAlignment = StringAlignment.Center
            };
            e.Graphics.DrawString("Không có dữ liệu trong khoảng thời gian đã chọn", _fontEmptyGrid, brush, rect, sf);
        };
    }

    // Xử lý khi bấm nút Đăng nhập
    private async void btnDangNhap_Click(object? sender, EventArgs e)
    {
        string taiKhoan = txtEmail.Text.Trim();
        string matKhau = txtMatKhau.Text;

        if (string.IsNullOrEmpty(taiKhoan) || string.IsNullOrEmpty(matKhau))
        {
            MessageBox.Show("Vui lòng nhập đầy đủ Email/Mã và Mật khẩu!",
                "Thiếu thông tin", MessageBoxButtons.OK, MessageBoxIcon.Warning);
            txtEmail.Focus();
            return;
        }

        using var db = new AppDbContext();
        string nhap = taiKhoan.ToLowerInvariant();

        
        if (rdoDocGia.Checked)
            PhienDangNhap.DongBoTaiKhoanDocGia(db);

        TaiKhoan? tkTimThay;

        // TRƯỜNG HỢP 1: Đăng nhập vai trò Thủ thư / Quản trị
        if (rdoThuThu.Checked)
        {
            string emailThuThu = nhap.Contains('@') ? nhap : $"{nhap}@hcmuaf.edu.vn";

            tkTimThay = await db.TaiKhoans.FirstOrDefaultAsync(t =>
                    t.VaiTro == LoaiTaiKhoan.ThuThu &&
                    (t.TenDangNhap.ToLower() == nhap || t.TenDangNhap.ToLower() == emailThuThu));
        }
        // TRƯỜNG HỢP 2: Đăng nhập vai trò Độc giả / Sinh viên
        else
        {
            bool laEmail = nhap.Contains('@');
            string maDocGia = nhap.ToUpperInvariant();

            tkTimThay = await db.TaiKhoans.FirstOrDefaultAsync(t =>
                    t.VaiTro == LoaiTaiKhoan.DocGia &&
                    (laEmail
                        ? t.TenDangNhap.ToLower() == nhap
                        : t.MaDocGia == maDocGia));
        }


        // Nếu không tìm thấy tài khoản HOẶC mật khẩu nhập vào không khớp
        if (tkTimThay == null || !MatKhau.Kiem(matKhau, tkTimThay.MatKhauHash))
        {

            string thongBao = "Sai tài khoản, mật khẩu hoặc chọn sai vai trò!";
#if DEBUG
            string goiY = rdoThuThu.Checked
                ? "Tài khoản Thủ thư mẫu: admin@hcmuaf.edu.vn / Mật khẩu: 123"
                : "Tài khoản Sinh viên mẫu: 24130265@st.hcmuaf.edu.vn (hoặc 24130265) / Mật khẩu: 123";
            thongBao += $"\n\n({goiY})";
#endif
            MessageBox.Show(thongBao, "Đăng nhập thất bại", MessageBoxButtons.OK, MessageBoxIcon.Error);
            return;
        }

            // Tự động nâng cấp mật khẩu từ chuẩn SHA256 cũ lên PBKDF2 mới
            if (MatKhau.CanNangCap(tkTimThay.MatKhauHash))
        {
            tkTimThay.MatKhauHash = MatKhau.Bam(matKhau);
            await db.SaveChangesAsync();
        }
        if (tkTimThay.BatBuocDoiMatKhau)
        {
            using var dlg = new DlgDoiMatKhau(tkTimThay.MaTaiKhoan);
            if (dlg.ShowDialog(this) != DialogResult.OK)
                return;                                   

            await db.Entry(tkTimThay).ReloadAsync();  // Tải lại dữ liệu tài khoản từ CSDL sau khi đổi mật khẩu

    // ===================== LÆ¯á»I VĂ€ HĂ€M PHá»¤ =====================

    private async Task NapDanhSachAsync(string? maCanChon)
    {
        using var db = new AppDbContext();
        var ds = await db.DocGias
            .AsNoTracking()
            .OrderBy(d => d.MaDocGia)
            .ToListAsync();

        _ds.RaiseListChangedEvents = false;
        _ds.Clear();
        foreach (var d in ds)
            _ds.Add(new DocGiaDto(d.MaDocGia, d.HoTen, d.SoDienThoai, d.BiKhoaThe, d.TienNoPhat));
        _ds.RaiseListChangedEvents = true;
        _ds.ResetBindings();

        if (maCanChon != null && ChonDong(maCanChon))
            return;

        BoChonVaLamMoiKhungNhap();
    }

    // Láº¥y dĂ²ng Ä‘ang Ä‘Æ°á»£c chá»n tháº­t sá»± (null náº¿u chÆ°a chá»n dĂ²ng nĂ o)
    private DocGiaDto? LayDongChon()
    {
        if (dgvDocGia.SelectedRows.Count == 0) return null;
        return dgvDocGia.SelectedRows[0].DataBoundItem as DocGiaDto;
    }

    private bool ChonDong(string ma)
    {
        foreach (DataGridViewRow dong in dgvDocGia.Rows)
        {
            if (dong.DataBoundItem is DocGiaDto x && x.MaDocGia == ma)
            {
                dgvDocGia.ClearSelection();
                dong.Selected = true;
                dgvDocGia.CurrentCell = dong.Cells["MaDocGia"];
                return true;
            }
        }
        return false;
    }

    private void BoChonVaLamMoiKhungNhap()
    {

        dgvDocGia.ClearSelection();
        _dangLamMoi = false;

        txtMaDocGia.ReadOnly = false;
        txtMaDocGia.Clear();
        txtHoTen.Clear();
        txtSoDienThoai.Clear();
        nudSoTienTra.Value = 0;
        lblTienNoPhat.Text = "Ná»£ pháº¡t: 0 VNÄ";
        btnKhoa.Text = "KhĂ³a tháº»";
        errProvider.Clear();
    }

    private void CauHinhLuoi()
    {
        dgvDocGia.ReadOnly = true;

        dgvDocGia.AllowUserToDeleteRows = false;

            c5.HeaderText = "Ná»£ pháº¡t";

        }
    }

    // Lấy thể loại được chọn và toàn bộ thể loại con cháu của nó
    private HashSet<string> LayTheLoaiVaCon(string ma)
    {
        var ketQua = new HashSet<string> { ma };
        var hangDoi = new Queue<string>();
        hangDoi.Enqueue(ma);

        while (hangDoi.Count > 0)
        {
            string cha = hangDoi.Dequeue();
            foreach (var con in _tatCaTheLoai.Where(t => t.MaTheLoaiCha == cha))
            {
                if (ketQua.Add(con.MaTheLoai))
                    hangDoi.Enqueue(con.MaTheLoai);
            }
        }
        return ketQua;
    }

    private async Task NapSachAsync()
    {
        using var db = new AppDbContext();
        var tatCa = await db.DauSachs.AsNoTracking()
            .Include(d => d.TheLoai)
            .Include(d => d.CuonSachs)
            .ToListAsync();

        string tuKhoa = txtTimKiem.Text.Trim();
        string maTheLoai = cboTheLoai.SelectedValue as string ?? "";
        HashSet<string>? tapTheLoai = maTheLoai.Length > 0 ? LayTheLoaiVaCon(maTheLoai) : null;

        // Lọc trong bộ nhớ để tìm không phân biệt hoa thường kể cả tiếng Việt có dấu
        var ds = tatCa
            .Where(d => tapTheLoai == null || tapTheLoai.Contains(d.MaTheLoai))
            .Where(d => tuKhoa.Length == 0
                     || d.TenDauSach.Contains(tuKhoa, StringComparison.CurrentCultureIgnoreCase)
                     || d.TacGia.Contains(tuKhoa, StringComparison.CurrentCultureIgnoreCase)
                     || d.MaDauSach.Contains(tuKhoa, StringComparison.CurrentCultureIgnoreCase))
            .OrderBy(d => d.MaDauSach)
            .Select(d => new DauSachDto(
                d.MaDauSach, d.TenDauSach, d.TacGia, d.NamXuatBan, d.GiaBia,
                d.MaTheLoai, d.TheLoai?.TenTheLoai ?? "",
                d.CuonSachs.Count,
                d.CuonSachs.Count(c => c.TrangThai == TrangThaiSach.SanSang)))
            .ToList();

        dgvSachThuVien.DataSource = ds;
        CapNhatSachDangChon();
    }

                return;

                y.MaDocGia == _maDocGia &&
                y.MaDauSach == sach.MaDauSach &&
                y.TrangThai == TrangThaiYeuCau.ChoDuyet);

                return;

            {
                MaDocGia = _maDocGia,
                MaDauSach = sach.MaDauSach,
                NgayGui = DateTime.Now,
                TrangThai = TrangThaiYeuCau.ChoDuyet
            });

            Toast.ThanhCong(this, $"Đã gửi yêu cầu mượn \"{sach.TenDauSach}\". Hãy nhận sách tại quầy khi được duyệt.");
        }
        catch (Exception ex)
        {
            NhatKy.Ghi(ex);
            Toast.Loi(this, "Không thể gửi yêu cầu. Vui lòng thử lại.");
        }
    }

    private async Task NapLichSuAsync()
    {
        using var db = new AppDbContext();
        DateTime homNay = DateTime.Today;

        // 1. Lịch sử mượn (mọi truy vấn đều lọc theo độc giả đang đăng nhập)
        var chiTiet = await db.ChiTietPhieuMuons.AsNoTracking()
            .Include(c => c.PhieuMuon)
            .Include(c => c.CuonSach).ThenInclude(cs => cs!.DauSach)
            .Where(c => c.PhieuMuon != null && c.PhieuMuon.MaDocGia == _maDocGia)
            .ToListAsync();

        dgvLichSuMuon.DataSource = chiTiet
            .OrderByDescending(c => c.PhieuMuon!.NgayMuon)
            .ThenByDescending(c => c.Id)
            .Select(c =>
            {
                var phieu = c.PhieuMuon!;
                string trangThai = c.NgayTraThucTe != null
                    ? "Đã trả"
                    : (homNay > phieu.NgayPhaiTra.Date ? "Quá hạn" : "Đang mượn");
                return new DongLichSu(
                    c.MaPhieuMuon,
                    c.CuonSach?.DauSach?.TenDauSach ?? "",
                    c.MaCuonSach,
                    phieu.NgayMuon.ToString("dd/MM/yyyy"),
                    phieu.NgayPhaiTra.ToString("dd/MM/yyyy"),
                    c.NgayTraThucTe?.ToString("dd/MM/yyyy") ?? "",
                    trangThai);
            })
            .ToList();

        // 2. Yêu cầu mượn đã gửi
        var yeuCau = await db.YeuCauMuons.AsNoTracking()
            .Include(y => y.DauSach)
            .Where(y => y.MaDocGia == _maDocGia)
            .ToListAsync();

        dgvYeuCauCuaToi.DataSource = yeuCau
            .OrderByDescending(y => y.NgayGui)
            .Select(y => new DongYeuCau(
                y.MaYeuCau,
                y.DauSach?.TenDauSach ?? y.MaDauSach,
                y.NgayGui.ToString("dd/MM/yyyy HH:mm"),
                y.TrangThai))
            .ToList();

        // 3. Hóa đơn phạt
        var hoaDon = await db.HoaDonPhats.AsNoTracking()
            .Where(h => h.MaDocGia == _maDocGia)
            .ToListAsync();

        dgvHoaDon.DataSource = hoaDon
            .OrderByDescending(h => h.NgayLap)
            .Select(h => new DongHoaDon(
                h.MaHoaDon,
                h.NgayLap.ToString("dd/MM/yyyy"),
                h.LyDo ?? "",
                (h.TienPhatQuaHan + h.TienBoiThuong).ToString("N0") + " đ",
                h.DaThanhToan ? "Đã thanh toán" : "Chưa thanh toán"))
            .ToList();

        await NapThongTinDocGiaAsync();   // làm mới số nợ ở tiêu đề
    }

    private async void btnLapPhieu_Click(object? sender, EventArgs e)
    {
        btnLapPhieu.Enabled = false;
        try
        {
            if (_maDocGia == null || _gio.Count == 0)
            {
                Toast.ThongTin(this, "Giá» mÆ°á»£n Ä‘ang trá»‘ng.");
                return;
            }

            var ngayMuon = dtpNgayMuon.Value.Date;
            if (ngayMuon > DateTime.Today)
            {
                Toast.CanhBao(this, "NgĂ y mÆ°á»£n khĂ´ng Ä‘Æ°á»£c lá»›n hÆ¡n ngĂ y hiá»‡n táº¡i.");
                return;
            }

            using var db = new AppDbContext();

            // Kiá»ƒm tra láº¡i vá»›i CSDL báº±ng cĂ¹ng quy táº¯c: trĂ¡nh xung Ä‘á»™t náº¿u mĂ n hĂ¬nh khĂ¡c vá»«a cáº­p nháº­t
            var kq = await MuonService.KiemTraAsync(db, _maDocGia, _gio.Count);
            if (!kq.Du)
            {
                Toast.CanhBao(this, "KhĂ´ng thá»ƒ láº­p phiáº¿u. " + kq.MoTa);
                return;
            }

            var dsMa = _gio.Select(g => g.MaCuonSach).ToList();
            var cacCuon = await db.CuonSachs.Where(c => dsMa.Contains(c.MaCuonSach)).ToListAsync();

            if (cacCuon.Count != dsMa.Count || cacCuon.Any(c => c.TrangThai != TrangThaiSach.SanSang))
            {
                Toast.CanhBao(this, "CĂ³ cuá»‘n trong giá» khĂ´ng cĂ²n á»Ÿ tráº¡ng thĂ¡i \"Sáºµn sĂ ng\". HĂ£y lĂ m má»›i vĂ  chá»n láº¡i.");
                return;
            }

            var phieu = new PhieuMuon
            {
                MaDocGia = _maDocGia,
                NgayMuon = ngayMuon,
                NgayPhaiTra = ngayMuon.AddDays(SoNgayMuon)
            };

            foreach (var cuon in cacCuon)
            {
                phieu.ChiTietPhieuMuons.Add(new ChiTietPhieuMuon
                {
                    MaCuonSach = cuon.MaCuonSach,
                    NgayTraThucTe = null
                });
                cuon.TrangThai = TrangThaiSach.DangMuon;
            }

            // Phiáº¿u xuáº¥t phĂ¡t tá»« yĂªu cáº§u online: Ä‘á»•i "ÄĂ£ duyá»‡t" trong CĂ™NG láº§n lÆ°u vá»›i phiáº¿u mÆ°á»£n.
            // Chá»‰ Ä‘á»•i khi cuá»‘n trong giá» Ä‘Ăºng lĂ  Ä‘áº§u sĂ¡ch mĂ  Ä‘á»™c giáº£ Ä‘Ă£ yĂªu cáº§u.
            if (_maYeuCauDangXuLy is int maYc)
            {
                var yc = await db.YeuCauMuons.FirstOrDefaultAsync(y => y.MaYeuCau == maYc);
                if (yc != null
                    && yc.TrangThai == TrangThaiYeuCau.ChoDuyet
                    && yc.MaDocGia == _maDocGia
                    && cacCuon.Any(c => c.MaDauSach == yc.MaDauSach))
                {
                    yc.TrangThai = TrangThaiYeuCau.DaDuyet;
                }
            }

            db.PhieuMuons.Add(phieu);
            await db.SaveChangesAsync();

            Toast.ThanhCong(this, $"Láº­p phiáº¿u mÆ°á»£n thĂ nh cĂ´ng! MĂ£ phiáº¿u: {phieu.MaPhieuMuon}");

            LamMoi();
        }
        catch (DbUpdateConcurrencyException)
        {
            Toast.CanhBao(this, "CĂ³ cuá»‘n vá»«a Ä‘Æ°á»£c cho mÆ°á»£n hoáº·c Ä‘á»•i tráº¡ng thĂ¡i á»Ÿ nÆ¡i khĂ¡c. HĂ£y lĂ m má»›i vĂ  chá»n láº¡i.");
        }
        catch (Exception ex)
        {
            BaoLoi(ex);
        }
        finally
        {
            CapNhatDem();
        }
    }

    private async Task KiemTraDocGiaAsync()

    private async Task NapCayTheLoaiAsync(string? maCanChon)
    {
        using (var db = new AppDbContext())
        {
            _tatCaTheLoai = await db.TheLoais.AsNoTracking().ToListAsync();
        }

        // Cập nhật ComboBox
        cboTheLoai.DataSource = null;
        cboTheLoai.DisplayMember = nameof(TheLoai.TenTheLoai);
        cboTheLoai.ValueMember = nameof(TheLoai.MaTheLoai);
        cboTheLoai.DataSource = _tatCaTheLoai.OrderBy(t => t.TenTheLoai).ToList();

        tvTheLoai.BeginUpdate();
        tvTheLoai.Nodes.Clear();

        var nutGoc = new TreeNode("Tất cả sách") { Tag = null };
        tvTheLoai.Nodes.Add(nutGoc);
        NapNutCon(nutGoc, null);

        tvTheLoai.ExpandAll();
        tvTheLoai.EndUpdate();

        // Chọn nút
        TreeNode? nutChon = maCanChon == null ? null : TimNut(nutGoc, maCanChon);
        tvTheLoai.SelectedNode = nutChon ?? nutGoc;
    }

    // ĐỆ QUY: tìm các thể loại con trực tiếp của maCha rồi gọi lại chính nó cho từng con
    private void NapNutCon(TreeNode nutCha, string? maCha)
    {
        var cacCon = _tatCaTheLoai
            .Where(t => t.MaTheLoaiCha == maCha)
            .OrderBy(t => t.TenTheLoai);

        foreach (var tl in cacCon)
        {
            var nutCon = new TreeNode(tl.TenTheLoai) { Tag = tl.MaTheLoai };
            nutCha.Nodes.Add(nutCon);
            NapNutCon(nutCon, tl.MaTheLoai);
        }
    }

    //de quy
    private static TreeNode? TimNut(TreeNode nut, string ma)
    {
        foreach (TreeNode con in nut.Nodes)
        {
            if (con.Tag as string == ma) return con;
            var kq = TimNut(con, ma);
            if (kq != null) return kq;
        }
        return null;
    }

    //layma
    private List<string> LayMaTheLoaiCon(string ma)
    {
        var ketQua = new HashSet<string> { ma };
        var hangDoi = new Queue<string>();
        hangDoi.Enqueue(ma);

        while (hangDoi.Count > 0)
        {
            string cha = hangDoi.Dequeue();
            foreach (var con in _tatCaTheLoai.Where(t => t.MaTheLoaiCha == cha))
            {
                if (ketQua.Add(con.MaTheLoai))      
                    hangDoi.Enqueue(con.MaTheLoai);
            }
        }
        return ketQua.ToList();
    }

    private async Task NapDauSachAsync(string? maTheLoai)
    {
        using var db = new AppDbContext();

        IQueryable<DauSach> truyVan = db.DauSachs
            .AsNoTracking()
            .Include(d => d.CuonSachs)
            .Include(d => d.TheLoai);

        if (maTheLoai != null)
        {
            var dsMa = LayMaTheLoaiCon(maTheLoai);
            truyVan = truyVan.Where(d => dsMa.Contains(d.MaTheLoai));
        }

        var ds = await truyVan.OrderBy(d => d.MaDauSach).ToListAsync();

        _dsDauSach.RaiseListChangedEvents = false;
        _dsDauSach.Clear();
        foreach (var d in ds)
        {
            _dsDauSach.Add(new DauSachDto(
                d.MaDauSach,
                d.TenDauSach,
                d.TacGia,
                d.NamXuatBan,
                d.GiaBia,
                d.MaTheLoai,
                d.TheLoai?.TenTheLoai ?? "",
                d.CuonSachs.Count,
                d.CuonSachs.Count(c => c.TrangThai == TrangThaiSach.SanSang)));
        }
        _dsDauSach.RaiseListChangedEvents = true;
        _dsDauSach.ResetBindings();

        if (_maCanChon != null && ChonDongDauSach(_maCanChon))
            _maCanChon = null;

    }

    private async Task NapCuonSachAsync(string maDauSach)
    {
        using var db = new AppDbContext();
        var ds = await db.CuonSachs
            .AsNoTracking()
            .Where(c => c.MaDauSach == maDauSach)
            .OrderBy(c => c.MaCuonSach.Length)
            .ThenBy(c => c.MaCuonSach)
            .ToListAsync();

        _dsCuonSach.Clear();
        foreach (var c in ds)
            _dsCuonSach.Add(new CuonSachDto(c.MaCuonSach, c.ViTriKe, c.TrangThai));
    }

    // Tách phần số sau dấu "_" của mỗi mã (SACH01_12 thì lấy 12) rồi tìm số lớn nhất
    private static int LaySoLonNhat(IEnumerable<string> danhSachMa)
    {
        int lonNhat = 0;
        foreach (var ma in danhSachMa)
        {
            string phanSo = ma[(ma.LastIndexOf('_') + 1)..];
            if (int.TryParse(phanSo, out int so) && so > lonNhat)
                lonNhat = so;
        }
        return lonNhat;
    }

    // Chọn lại một dòng đầu sách trong lưới theo mã
    private bool ChonDongDauSach(string ma)
    {
        foreach (DataGridViewRow dong in dgvDauSach.Rows)
        {
            if (dong.DataBoundItem is DauSachDto x && x.MaDauSach == ma)
            {
                dgvDauSach.CurrentCell = dong.Cells["MaDauSach"];
                dong.Selected = true;
                return true;
            }
        }
        return false;
    }

            string maDauSach = cuon.MaDauSach;
            db.CuonSachs.Remove(cuon);
            await db.SaveChangesAsync();

            _maCanChon = maDauSach;
            await NapDauSachAsync(tvTheLoai.SelectedNode?.Tag as string);
            Toast.ThanhCong(this, $"Đã xóa cuốn {ma}.");
        }
        catch (Exception ex)
        {
            BaoLoi(ex);
        }
    }
    private static void BaoLoi(Exception ex)
    {
        MessageBox.Show($"Có lỗi xảy ra:\n{ex.Message}", "Lỗi",
            MessageBoxButtons.OK, MessageBoxIcon.Error);
    }

    private void TaiDanhSachPhieuDangMuon(string tuKhoa = "")
    {
        using var db = new AppDbContext();

        var query = db.PhieuMuons
            .AsNoTracking()
            .Where(p => p.ChiTietPhieuMuons.Any(ct => ct.NgayTraThucTe == null));

        if (!string.IsNullOrWhiteSpace(tuKhoa))
        {
            tuKhoa = tuKhoa.Trim().ToLower();
            query = query.Where(p =>
                p.MaPhieuMuon.ToString().Contains(tuKhoa) ||
                (p.DocGia != null && p.DocGia.HoTen.ToLower().Contains(tuKhoa)));
        }

        var danhSach = query
            .OrderByDescending(p => p.NgayMuon)
            .Select(p => new
            {
                Id = p.MaPhieuMuon,
                p.MaDocGia,
                TenDocGia = p.DocGia != null ? p.DocGia.HoTen : "",
                p.NgayMuon
            })
            .ToList()
            .Select(p => new
            {
                p.Id,
                p.MaDocGia,
                HienThi = $"Phiếu #{p.Id} - {p.TenDocGia} (Mượn: {p.NgayMuon:dd/MM/yyyy})"
            })
            .ToList();

        cboPhieuMuon.DisplayMember = "HienThi";
        cboPhieuMuon.ValueMember = "Id";
        cboPhieuMuon.DataSource = danhSach;

        if (danhSach.Count == 0)
        {
            dgvChiTietMuon.Rows.Clear();
            TinhTongTienPhat();
            TaiHoaDonPhatCuaDocGia();
        }

    private void TinhTongTienPhat()
    {
        decimal donGiaTre = numDonGiaTre.Value;
        decimal mucPhatHuHong = numTienPhatHuHong.Value;

        int tongNgayTre = 0;
        decimal phatHuHong = 0;
        var lyDoList = new List<string>();

        foreach (DataGridViewRow row in dgvChiTietMuon.Rows)
        {
            bool duocChon = Convert.ToBoolean(row.Cells["ChonTra"].Value ?? false);
            if (!duocChon) continue;

            int ngayTre = Convert.ToInt32(row.Cells["SoNgayTre"].Value ?? 0);
            string tinhTrang = row.Cells["TinhTrangTra"].Value?.ToString() ?? TinhTrang.BinhThuong;
            string tenSach = row.Cells["TenDauSach"].Value?.ToString() ?? "";

            if (ngayTre > 0)
            {
                tongNgayTre += ngayTre;
                lyDoList.Add($"Trễ {ngayTre} ngày ({tenSach})");
            }

            if (tinhTrang == TinhTrang.HuHong || tinhTrang == TinhTrang.MatSach)
            {
                phatHuHong += mucPhatHuHong;
                lyDoList.Add($"{tinhTrang} ({tenSach})");
            }
        }

        _coSachTre = tongNgayTre > 0;
        _tienPhatTre = tongNgayTre * donGiaTre;
        _tienPhatHuHong = phatHuHong;
        _tongTienPhat = _tienPhatTre + _tienPhatHuHong;

        lblTongTienPhat.Text = $"{_tongTienPhat:N0} VNĐ";
        txtLyDoPhat.Text = string.Join("; ", lyDoList);
    }

    private async void btnXacNhanTra_Click(object? sender, EventArgs e)
    {
        try
        {
            if (cboPhieuMuon.SelectedValue is not int phieuMuonId)
            {
                Toast.CanhBao(this, "Vui lòng chọn phiếu mượn cần trả!");
                return;
            }

            var chiTietDuocChon = new Dictionary<int, string>();
            foreach (DataGridViewRow row in dgvChiTietMuon.Rows)
            {
                if (!Convert.ToBoolean(row.Cells["ChonTra"].Value ?? false)) continue;
                int ctId = Convert.ToInt32(row.Cells["ChiTietId"].Value);
                chiTietDuocChon[ctId] = row.Cells["TinhTrangTra"].Value?.ToString() ?? TinhTrang.BinhThuong;
            }

            if (chiTietDuocChon.Count == 0)
            {
                Toast.CanhBao(this, "Vui lòng tích chọn ít nhất một cuốn sách để trả!");
                return;
            }

            // Tính lại từ lưới ngay lúc xác nhận, không dùng số cũ
            TinhTongTienPhat();

            if (_coSachTre && numDonGiaTre.Value == 0)
            {
                var hoi = MessageBox.Show(
                    "Có sách trả trễ hạn nhưng \"Đơn giá phạt trễ\" đang là 0, nên sẽ không bị phạt.\n" +
                    "Vẫn tiếp tục xác nhận trả sách?",
                    "Chưa tính phạt trễ", MessageBoxButtons.YesNo, MessageBoxIcon.Question);
                if (hoi != DialogResult.Yes) { numDonGiaTre.Focus(); return; }
            }

            using var db = new AppDbContext();
            var phieu = await db.PhieuMuons
                .Include(p => p.ChiTietPhieuMuons)
                    .ThenInclude(ct => ct.CuonSach)
                .FirstOrDefaultAsync(p => p.MaPhieuMuon == phieuMuonId);

            if (phieu == null)
            {
                Toast.CanhBao(this, "Không tìm thấy thông tin phiếu mượn!");
                return;
            }

            int soCuonTra = 0;
            foreach (var ct in phieu.ChiTietPhieuMuons)
            {
                // Bỏ qua cuốn không được chọn, hoặc đã được trả ở nơi khác
                if (!chiTietDuocChon.TryGetValue(ct.Id, out string? tinhTrang) || ct.NgayTraThucTe != null)
                    continue;

                ct.NgayTraThucTe = DateTime.Now;
                ct.TinhTrangKhiTra = tinhTrang;
                soCuonTra++;

                if (ct.CuonSach != null)
                {
                    ct.CuonSach.TrangThai = tinhTrang switch
                    {
                        TinhTrang.HuHong => TrangThaiSach.Hong,
                        TinhTrang.MatSach => TrangThaiSach.Mat,
                        _ => TrangThaiSach.SanSang
                    };
                }
            }

            if (soCuonTra == 0)
            {
                Toast.CanhBao(this, "Các cuốn đã chọn đã được trả trước đó.");
                TaiDanhSachPhieuDangMuon(txtTimKiem.Text);
                return;
            }

            // Hóa đơn + nợ phạt đi qua PhatService, lưu chung một lần với việc trả sách
            await PhatService.TaoHoaDonAsync(db, phieu.MaDocGia, phieu.MaPhieuMuon,
                _tienPhatTre, _tienPhatHuHong, txtLyDoPhat.Text, chkDaThanhToanNgay.Checked);

            await db.SaveChangesAsync();

            Toast.ThanhCong(this, "Đã ghi nhận trả sách thành công!");
            TaiDanhSachPhieuDangMuon(txtTimKiem.Text);
        }
        catch (Exception ex)
        {
            System.Diagnostics.Debug.WriteLine(ex);
            Toast.Loi(this, "Không thể ghi nhận trả sách. Vui lòng thử lại.");
        }
    }

    private void TaiHoaDonPhatCuaDocGia(string? _ = null)

    private static Font? TaoFontIcon()
    {
        foreach (var ten in new[] { "Segoe Fluent Icons", "Segoe MDL2 Assets" })
        {
            if (FontFamily.Families.Any(f => f.Name == ten))
                return new Font(ten, 11F, FontStyle.Regular);
        }

    // =====================================================================

    public static void Apply(Control root)
    {
        if (root is Form form)
        {
            form.BackColor = BgMain;
            form.Font = FontNormal;
            form.ForeColor = TextDark;
        }

        ApDungDeQuy(root);
    }

    private static void ApDungDeQuy(Control parent)
    {
        foreach (Control c in parent.Controls)
        {
            switch (c)
            {
                case DataGridView dgv:
                    StyleGrid(dgv);
                    break;

                case Button btn:
                    StyleButtonAuto(btn);
                    break;

                case GroupBox grp:
                    StyleGroup(grp);
                    break;

                case TreeView tv:
                    StyleTree(tv);
                    break;

                case TextBox tb:
                    StyleTextBox(tb);
                    break;

                case SplitContainer spl:
                    StyleSplit(spl);
                    break;

                case ComboBox or NumericUpDown or DateTimePicker:
                    c.Font = FontNormal;
                    c.ForeColor = TextDark;
                    break;

                case CheckBox or RadioButton:
                    // GroupBox dùng chữ đậm, nên phải đặt lại để không "thừa kế" chữ đậm
                    c.Font = FontNormal;
                    c.ForeColor = TextDark;
                    break;

                case Label lbl:
                    lbl.Font = FontNormal;
                    lbl.ForeColor = TextDark;
                    break;
            }

            // Không đi sâu vào các control có control con nội bộ (ô gõ bên trong NumericUpDown, thanh cuộn...)
            if (c.HasChildren && c is not (NumericUpDown or ComboBox or DateTimePicker or DataGridView or TreeView or ListView))
            {
                ApDungDeQuy(c);
            }
        }
    }

    private sealed class NutState
    {
        public Color Mau = Color.Empty;
        public bool Hover, Down, Sidebar, Active;
        public string Icon = "";

    private static readonly ConditionalWeakTable<Button, NutState> _nut = new();

    private static NutState LayTrangThai(Button btn)
    {
        if (_nut.TryGetValue(btn, out var st)) return st;

        st = new NutState();
        _nut.Add(btn, st);
        var state = st;

        btn.MouseEnter += (s, e) => { state.Hover = true; btn.Invalidate(); };
        btn.MouseLeave += (s, e) => { state.Hover = false; state.Down = false; btn.Invalidate(); };
        btn.MouseDown += (s, e) => { if (e.Button == MouseButtons.Left) { state.Down = true; btn.Invalidate(); } };
        btn.MouseUp += (s, e) => { state.Down = false; btn.Invalidate(); };
        btn.EnabledChanged += (s, e) => btn.Invalidate();
        btn.Paint += (s, e) => VeNut(btn, state, e.Graphics);
        return st;
    }

    public static void StyleButtonAuto(Button btn)

    private static Color Mix(Color a, Color b, float t) =>
        Color.FromArgb(a.R + (int)((b.R - a.R) * t), a.G + (int)((b.G - a.G) * t), a.B + (int)((b.B - a.B) * t));

    private static GraphicsPath TaoBoGoc(Rectangle r, int bk)
    {
        var p = new GraphicsPath();
        if (bk <= 0)
        {
            p.AddRectangle(r);
            return p;
        }
        int d = bk * 2;
        p.AddArc(r.X, r.Y, d, d, 180, 90);
        p.AddArc(r.Right - d, r.Y, d, d, 270, 90);
        p.AddArc(r.Right - d, r.Bottom - d, d, d, 0, 90);
        p.AddArc(r.X, r.Bottom - d, d, d, 90, 90);
        p.CloseFigure();
        return p;
    }

    private static void VeNut(Button btn, NutState st, Graphics g)
    {
        g.SmoothingMode = SmoothingMode.AntiAlias;
        g.Clear(btn.Parent?.BackColor ?? BgMain);   // tô nền góc bo bằng màu của vùng chứa

        Color chu = btn.Enabled ? btn.ForeColor : Color.FromArgb(148, 163, 184);
        Color nen;
        if (!btn.Enabled) nen = Color.FromArgb(226, 232, 240);
        else if (st.Sidebar) nen = st.Active ? Primary : (st.Hover ? Mix(PrimaryDark, Color.White, 0.12f) : PrimaryDark);
        else if (st.Down) nen = Mix(st.Mau, Color.Black, 0.18f);
        else if (st.Hover) nen = Mix(st.Mau, Color.White, 0.16f);
        else nen = st.Mau;

        Rectangle khung = st.Sidebar
            ? new Rectangle(0, 0, btn.Width, btn.Height)
            : new Rectangle(0, 0, btn.Width - 1, btn.Height - 1);

        using (var path = TaoBoGoc(khung, st.Sidebar ? 0 : 8))
        using (var brush = new SolidBrush(nen))
        {
            g.FillPath(brush, path);
        }

        var flags = TextFormatFlags.NoPadding | TextFormatFlags.HidePrefix
                  | TextFormatFlags.SingleLine | TextFormatFlags.VerticalCenter;

        // --- Nút sidebar: chữ canh trái + vạch nhấn khi đang chọn
        if (st.Sidebar)
        {
            if (st.Active)
            {
                using var vach = new SolidBrush(Accent);
                g.FillRectangle(vach, 0, 8, 4, btn.Height - 16);
            }
            var r = new Rectangle(20, 0, btn.Width - 28, btn.Height);
            TextRenderer.DrawText(g, btn.Text, btn.Font, r, chu, flags | TextFormatFlags.Left | TextFormatFlags.EndEllipsis);
            return;
        }

        // --- Nút thường: biểu tượng + chữ, canh giữa
        bool coIcon = DungIcon && FontIcon != null && st.Icon.Length > 0;
        int rongIcon = coIcon ? 24 : 0;
        int rongChu = TextRenderer.MeasureText(g, btn.Text, btn.Font, new Size(int.MaxValue, int.MaxValue), flags).Width;
        int kha = btn.Width - 16;
        if (coIcon && rongIcon + rongChu > kha)
        {
            // Không đủ chỗ cho cả icon lẫn chữ: bỏ icon để giữ nguyên chữ
            coIcon = false;
            rongIcon = 0;
        }
        int tong = rongIcon + rongChu;
        if (tong > kha)
        {
            rongChu = Math.Max(0, kha);
            tong = kha;
        }
        int x = (btn.Width - tong) / 2;

        if (coIcon)
        {
            TextRenderer.DrawText(g, st.Icon, FontIcon!, new Rectangle(x, 0, rongIcon, btn.Height),
                chu, flags | TextFormatFlags.HorizontalCenter);
        }
        TextRenderer.DrawText(g, btn.Text, btn.Font, new Rectangle(x + rongIcon, 0, rongChu + 2, btn.Height),
            chu, flags | TextFormatFlags.Left | TextFormatFlags.EndEllipsis);
    }

    private static void VeGroupBox(object? sender, PaintEventArgs e)
    {
        if (sender is not GroupBox grp) return;

        var g = e.Graphics;
        g.SmoothingMode = SmoothingMode.AntiAlias;
        Color cha = grp.Parent?.BackColor ?? BgMain;
        g.Clear(cha);

        int th = grp.Font.Height;
        var khung = new Rectangle(0, th / 2, grp.Width - 1, grp.Height - th / 2 - 1);
        using (var path = TaoBoGoc(khung, 8))
        using (var nen = new SolidBrush(CardBg))
        using (var vien = new Pen(Border))
        {
            g.FillPath(nen, path);
            g.DrawPath(vien, path);
        }

        if (!string.IsNullOrEmpty(grp.Text))
        {
            var flags = TextFormatFlags.NoPadding | TextFormatFlags.HidePrefix | TextFormatFlags.SingleLine;
            var size = TextRenderer.MeasureText(g, grp.Text, grp.Font, new Size(int.MaxValue, int.MaxValue), flags);
            using (var xoaVien = new SolidBrush(cha))
                g.FillRectangle(xoaVien, 10, 0, size.Width + 8, th);
            TextRenderer.DrawText(g, grp.Text, grp.Font, new Point(14, 0), grp.ForeColor, flags);
        }
    }

    public static void StyleTree(TreeView tv)
    {
        tv.BackColor = CardBg;
        tv.ForeColor = TextDark;
        tv.Font = FontNormal;
        tv.BorderStyle = BorderStyle.None;
        tv.ShowLines = false;
        tv.ItemHeight = 32;
        tv.Indent = 22;
        tv.HideSelection = false;
        tv.DrawMode = TreeViewDrawMode.OwnerDrawText;

        if (DangKyMotLan(tv))
        {
            tv.HandleCreated += (s, e) => SetWindowTheme(tv.Handle, "explorer", null);
            if (tv.IsHandleCreated) SetWindowTheme(tv.Handle, "explorer", null);

            tv.DrawNode += VeNutCay;

            tv.MouseMove += (s, e) =>
            {
                var moi = tv.GetNodeAt(e.X, e.Y);
                var st = _cayHover.GetOrCreateValue(tv);
                if (ReferenceEquals(st.Node, moi)) return;

                var cu = st.Node;
                st.Node = moi;
                if (cu != null && cu.TreeView == tv) tv.Invalidate(cu.Bounds);
                if (moi != null) tv.Invalidate(moi.Bounds);
            };
            tv.MouseLeave += (s, e) =>
            {
                var st = _cayHover.GetOrCreateValue(tv);
                var cu = st.Node;
                st.Node = null;
                if (cu != null && cu.TreeView == tv) tv.Invalidate(cu.Bounds);
            };
        }
    }

    private static void VeNutCay(object? sender, DrawTreeNodeEventArgs e)
    {
        if (sender is not TreeView tv || e.Node == null || e.Bounds.Width <= 0) return;

        bool chon = (e.State & TreeNodeStates.Selected) != 0;
        bool ditro = _cayHover.TryGetValue(tv, out var st) && ReferenceEquals(st.Node, e.Node);

        var g = e.Graphics;
        g.SmoothingMode = SmoothingMode.AntiAlias;
        var r = e.Bounds;

        using (var xoa = new SolidBrush(tv.BackColor))
            g.FillRectangle(xoa, r);

        if (chon || ditro)
        {
            var vien = new Rectangle(r.X, r.Y + 1, r.Width, r.Height - 2);
            using var path = TaoBoGoc(vien, 6);
            using var br = new SolidBrush(chon ? Primary : GridHover);
            g.FillPath(br, path);
        }

        TextRenderer.DrawText(g, e.Node.Text, e.Node.NodeFont ?? tv.Font, r,
            chon ? Color.White : TextDark,
            TextFormatFlags.VerticalCenter | TextFormatFlags.HorizontalCenter
            | TextFormatFlags.NoPrefix | TextFormatFlags.EndEllipsis);
    }

    // =====================================================================
    public static void FadeIn(Form f, int ms = 250)
    {
        if (!f.TopLevel) return;

        f.Opacity = 0;
        var timer = new System.Windows.Forms.Timer { Interval = 15 };
        double buoc = 1.0 / Math.Max(1, ms / 15);

        timer.Tick += (s, e) =>
        {
            if (f.IsDisposed)
            {
                timer.Stop();
                timer.Dispose();
                return;
            }
            f.Opacity = Math.Min(1.0, f.Opacity + buoc);
            if (f.Opacity >= 1.0)
            {
                timer.Stop();
                timer.Dispose();
            }
        };
        f.Shown += (s, e) => timer.Start();
    }

using System.Diagnostics;
using System.Drawing;

namespace QuanLyThuVien.UI;

public enum ToastLoai { ThanhCong, Loi, CanhBao, ThongTin }

/// <summary>
/// Thông báo nhỏ trượt vào góc phải trên của cửa sổ chính rồi tự biến mất.
/// Cách dùng (trong bất kỳ Form nào):  Toast.ThanhCong(this, "Đã lưu thành công");
/// </summary>
public sealed class Toast : Form
{
    // ===== API dùng ở các Form =====
    private static Toast? _dangHien;

    public static void ThanhCong(Control nguon, string noiDung) => Hien(nguon, noiDung, ToastLoai.ThanhCong, 2800);
    public static void Loi(Control nguon, string noiDung) => Hien(nguon, noiDung, ToastLoai.Loi, 4000);
    public static void CanhBao(Control nguon, string noiDung) => Hien(nguon, noiDung, ToastLoai.CanhBao, 3500);
    public static void ThongTin(Control nguon, string noiDung) => Hien(nguon, noiDung, ToastLoai.ThongTin, 2800);

    public static void Hien(Control nguon, string noiDung, ToastLoai loai = ToastLoai.ThanhCong, int ms = 2800)
    {
        // Form con nhúng trong FrmMain (TopLevel = false) -> lấy cửa sổ chính làm nơi hiển thị
        var host = nguon.TopLevelControl as Form ?? nguon.FindForm();

        if (host == null || host.IsDisposed || !host.IsHandleCreated || host.WindowState == FormWindowState.Minimized)
        {
            // Dự phòng: không có cửa sổ chính để gắn thì dùng hộp thoại thường
            MessageBox.Show(noiDung, "Thông báo", MessageBoxButtons.OK, MessageBoxIcon.Information);
            return;
        }

        _dangHien?.DongNgay();                 // chỉ giữ 1 thông báo tại một thời điểm
        var t = new Toast(noiDung, loai, ms);
        _dangHien = t;
        t.HienTai(host);
    }

    // ===== Thiết lập nội bộ =====
    private const TextFormatFlags FlagsChu = TextFormatFlags.Left | TextFormatFlags.Top
        | TextFormatFlags.WordBreak | TextFormatFlags.NoPrefix | TextFormatFlags.NoPadding;

    private const int ThoiGianVao = 260;   // ms
    private const int ThoiGianRa = 220;    // ms

    private readonly string _noiDung;
    private readonly string _tieuDe;
    private readonly ToastLoai _loai;
    private readonly Color _mau;
    private readonly int _thoiGianHien;

    private readonly System.Windows.Forms.Timer _timer = new() { Interval = 15 };
    private readonly Stopwatch _dongHo = new();

    private int _giaiDoan;          // 0 = đang trượt vào, 1 = đang hiển thị, 2 = đang trượt ra
    private int _xDich;
    private int _khoangTruot;
    private double _tiLe = 1.0;     // phần thời gian còn lại, dùng vẽ thanh tiến trình

    private readonly int _xChu;
    private readonly int _rongChu;
    private readonly int _yTieuDe;
    private readonly int _yNoiDung;

    private int P(int v) => (int)Math.Round(v * DeviceDpi / 96.0);   // đổi theo độ phóng đại màn hình

    private Toast(string noiDung, ToastLoai loai, int thoiGianHien)
    {
        _noiDung = noiDung;
        _loai = loai;
        _thoiGianHien = Math.Max(800, thoiGianHien);

        (_mau, _tieuDe) = loai switch
        {
            ToastLoai.ThanhCong => (Color.FromArgb(22, 163, 74), "Thành công"),
            ToastLoai.Loi => (Theme.Danger, "Có lỗi xảy ra"),
            ToastLoai.CanhBao => (Color.FromArgb(217, 119, 6), "Lưu ý"),
            _ => (Theme.Secondary, "Thông tin")
        };

        FormBorderStyle = FormBorderStyle.None;
        StartPosition = FormStartPosition.Manual;
        ShowInTaskbar = false;
        BackColor = Color.White;
        DoubleBuffered = true;
        Opacity = 0;
        Cursor = Cursors.Hand;

        // Tính kích thước theo độ dài nội dung
        int rongToast = P(360);
        _xChu = P(64);
        _rongChu = rongToast - _xChu - P(16);
        _yTieuDe = P(12);
        _yNoiDung = _yTieuDe + Theme.FontBold.Height + P(2);

        var kich = TextRenderer.MeasureText(noiDung, Theme.FontNormal, new Size(_rongChu, int.MaxValue), FlagsChu);
        ClientSize = new Size(rongToast, Math.Max(P(68), _yNoiDung + kich.Height + P(16)));

        _timer.Tick += Tick;
        Click += (s, e) => BatDauDong();   // bấm vào thông báo để tắt ngay
    }

    // Không cướp focus của cửa sổ đang làm việc; không hiện trên thanh taskbar; có bóng đổ
    protected override bool ShowWithoutActivation => true;

    protected override CreateParams CreateParams
    {
        get
        {
            var cp = base.CreateParams;
            cp.ExStyle |= 0x08000000 | 0x00000080;   // WS_EX_NOACTIVATE | WS_EX_TOOLWINDOW
            cp.ClassStyle |= 0x00020000;             // CS_DROPSHADOW
            return cp;
        }
    }

    [DllImport("dwmapi.dll")]
    private static extern int DwmSetWindowAttribute(IntPtr hwnd, int attr, ref int attrValue, int attrSize);

    protected override void OnHandleCreated(EventArgs e)
    {
        base.OnHandleCreated(e);
        try
        {
            // Windows 11: bo góc mượt cho cửa sổ không viền (Windows 10 bỏ qua, vẫn hoạt động bình thường)
            int bo = 2;   // DWMWCP_ROUND
            DwmSetWindowAttribute(Handle, 33, ref bo, sizeof(int));
        }
        catch { }
    }

    private void HienTai(Form host)
    {
        // Nếu host là hộp thoại nhỏ (như DlgDoiMatKhau) thì neo Toast theo cửa sổ cha
        var khung = (host.Owner != null && host.ClientSize.Width < Width + P(60))
            ? host.Owner
            : host;

        var goc = khung.PointToScreen(Point.Empty);
        _xDich = goc.X + khung.ClientSize.Width - Width - P(24);
        _khoangTruot = P(70);

        int leTren = khung == host ? P(76) : P(16);
        Location = new Point(_xDich + _khoangTruot, goc.Y + leTren);
        Show(host);

        _dongHo.Restart();
        _timer.Start();
    }

    // ===== Hiệu ứng =====
    private void Tick(object? sender, EventArgs e)
    {
        long t = _dongHo.ElapsedMilliseconds;

        if (_giaiDoan == 0)
        {
            double p = Math.Min(1.0, t / (double)ThoiGianVao);
            double ease = 1 - Math.Pow(1 - p, 3);            // chậm dần về cuối
            Left = _xDich + (int)((1 - ease) * _khoangTruot);
            Opacity = ease;
            if (p >= 1.0)
            {
                _giaiDoan = 1;
                _dongHo.Restart();
            }
        }
        else if (_giaiDoan == 1)
        {
            _tiLe = 1.0 - Math.Min(1.0, t / (double)_thoiGianHien);
            Invalidate(new Rectangle(0, Height - P(4), Width, P(4)));
            if (t >= _thoiGianHien) BatDauDong();
        }
        else
        {
            double p = Math.Min(1.0, t / (double)ThoiGianRa);
            double ease = p * p * p;                          // nhanh dần
            Left = _xDich + (int)(ease * _khoangTruot);
            Opacity = 1 - ease;
            if (p >= 1.0)
            {
                _timer.Stop();
                Close();
            }
        }
    }

    private void BatDauDong()
    {
        if (_giaiDoan == 2 || IsDisposed) return;
        _giaiDoan = 2;
        _tiLe = 0;
        _dongHo.Restart();
    }

    private void DongNgay()
    {
        if (IsDisposed) return;
        _timer.Stop();
        Close();
    }

    protected override void OnFormClosed(FormClosedEventArgs e)
    {
        _timer.Stop();
        _timer.Dispose();
        if (ReferenceEquals(_dangHien, this)) _dangHien = null;
        base.OnFormClosed(e);
    }

    // ===== Vẽ =====
    protected override void OnPaint(PaintEventArgs e)
    {
        base.OnPaint(e);
        var g = e.Graphics;
        g.SmoothingMode = SmoothingMode.AntiAlias;

        // Viền mảnh quanh thông báo
        using (var vien = new Pen(Theme.Border))
            g.DrawRectangle(vien, 0, 0, Width - 1, Height - 1);

        // Thanh màu bên trái
        using (var thanh = new SolidBrush(_mau))
            g.FillRectangle(thanh, 0, 0, P(5), Height);

        // Vòng tròn biểu tượng
        int d = P(30);
        var vongTron = new Rectangle(P(20), _yTieuDe, d, d);
        using (var nen = new SolidBrush(_mau))
            g.FillEllipse(nen, vongTron);
        VeKyHieu(g, vongTron);

        // Chữ
        TextRenderer.DrawText(g, _tieuDe, Theme.FontBold,
            new Rectangle(_xChu, _yTieuDe, _rongChu, Theme.FontBold.Height), Theme.TextDark,
            TextFormatFlags.Left | TextFormatFlags.Top | TextFormatFlags.NoPadding | TextFormatFlags.NoPrefix);

        TextRenderer.DrawText(g, _noiDung, Theme.FontNormal,
            new Rectangle(_xChu, _yNoiDung, _rongChu, Height - _yNoiDung), Theme.Neutral, FlagsChu);

        // Thanh thời gian còn lại ở đáy
        if (_giaiDoan < 2 && _tiLe > 0)
        {
            using var tg = new SolidBrush(Color.FromArgb(140, _mau));
            g.FillRectangle(tg, 0, Height - P(3), (int)(Width * _tiLe), P(3));
        }
    }

    private void VeKyHieu(Graphics g, Rectangle r)
    {
        float x = r.X, y = r.Y, d = r.Width;
        using var but = new Pen(Color.White, Math.Max(2f, d / 10f))
        {
            StartCap = LineCap.Round,
            EndCap = LineCap.Round,
            LineJoin = LineJoin.Round
        };

        switch (_loai)
        {
            case ToastLoai.ThanhCong:      // dấu tích
                g.DrawLines(but, new[]
                {
                    new PointF(x + d * 0.28f, y + d * 0.53f),
                    new PointF(x + d * 0.44f, y + d * 0.69f),
                    new PointF(x + d * 0.73f, y + d * 0.34f)
                });
                break;

            case ToastLoai.Loi:            // dấu X
                g.DrawLine(but, x + d * 0.33f, y + d * 0.33f, x + d * 0.67f, y + d * 0.67f);
                g.DrawLine(but, x + d * 0.67f, y + d * 0.33f, x + d * 0.33f, y + d * 0.67f);
                break;

            default:                       // "!" hoặc "i"
                TextRenderer.DrawText(g, _loai == ToastLoai.CanhBao ? "!" : "i", Theme.FontBold, r, Color.White,
                    TextFormatFlags.HorizontalCenter | TextFormatFlags.VerticalCenter | TextFormatFlags.NoPadding);
                break;
        }
    }
}
